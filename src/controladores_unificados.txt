import { Request, Response, NextFunction } from 'express';

// 1. Historial por rango de fechas y entidad (para reportes o filtros generales)
export const obtenerHistorialPorFiltro = async (req: Request, res: Response, next: NextFunction) => {
  try {
    const { entidad, fechaDesde, fechaHasta, accion } = req.query;
    const filtro: Record<string, any> = {};

    if (entidad) {
      filtro.entidad = entidad;
    }

    if (accion) {
      filtro.accion = accion;
    }

    // Filtro por rango de fechas sobre createdAt
    if (fechaDesde || fechaHasta) {
      filtro.createdAt = {};
      if (fechaDesde) {
        filtro.createdAt.$gte = new Date(fechaDesde as string);
      }
      if (fechaHasta) {
        // Fin del día para incluir los registros hasta las 23:59:59
        const hasta = new Date(fechaHasta as string);
        hasta.setHours(23, 59, 59, 999);
        filtro.createdAt.$lte = hasta;
      }
    }

    const logs = await AuditLogModel.find(filtro)
      .populate('usuario', 'nombre email role')
      .sort({ createdAt: -1 });

    res.json(logs);
  } catch (error) {
    next(error);
  }
};

// Obtengo codigo de barra y numero unico de aca
const getIdentifieres = (monto: number) => {
  const timestamp = Date.now().toString().slice(-6);
  const random = Math.floor(1000 + Math.random() * 9000);
  const numeroActa = `ACTA-2026-${timestamp}-${random}`;
  const codigoBarras = `0442026${timestamp}${random}${Math.round(monto).toString().padStart(8, '0')}`;
  return { numeroActa, codigoBarras };
};

// 1. POST /infracciones -> Labrar Acta (Exclusivo inspectores)
export const generarInfraccion = async (req: Request, res: Response, next: NextFunction) => {
  try {
    const { vehiculo, tipoFalta, lugar, monto, fechaHora } = req.body || {};

    if (!vehiculo || !isValidObjectId(vehiculo)) {
      return res.status(400).json({ error: 'Debe especificar un ID de vehículo válido' });
    }

    const vehiculoExiste = await VehiculoModel.findById(vehiculo);
    if (!vehiculoExiste) {
      return res.status(404).json({ error: 'El vehículo especificado no existe' });
    }

    if (!tipoFalta || tipoFalta.trim() === '') {
      return res.status(400).json({ error: 'El tipo de falta es obligatorio' });
    }

    if (!lugar || lugar.trim() === '') {
      return res.status(400).json({ error: 'El lugar de la infracción es obligatorio' });
    }

    if (monto === undefined || Number(monto) <= 0) {
      return res.status(400).json({ error: 'El monto debe ser mayor a 0' });
    }

    const { numeroActa, codigoBarras } = getIdentifieres(Number(monto));

    const nuevaInfraccion = await InfraccionModel.create({
      numeroActa,
      codigoBarras,
      vehiculo,
      tipoFalta: tipoFalta.trim(),
      lugar: lugar.trim(),
      monto: Number(monto),
      fechaHora: fechaHora ? new Date(fechaHora) : undefined,
      estado: 'pendiente',
      inspector: req.user?.id,
    });

    if (req.user?.id) {
      await registrarAuditoria({
        entidad: 'Infraccion',
        entidadId: nuevaInfraccion._id.toString(),
        accion: 'CREACION',
        usuarioId: req.user.id,
        datosNuevos: nuevaInfraccion.toObject(),
        motivo: `Labrado de acta N° ${numeroActa}`,
        ip: req.ip,
      });
    }

    const baseUrl = process.env.FRONTEND_URL || 'http://localhost:5173';
    const urlPago = `${baseUrl}/pagos/${nuevaInfraccion.numeroActa}`;

    res.status(201).json({
      message: 'Acta de infracción labrada exitosamente',
      infraccion: {
        ...nuevaInfraccion.toObject(),
        urlPago,
      },
    });
  } catch (error) {
    next(error);
  }
};

const ROLES_PERMITIDOS = ['inspector', 'administrativo', 'admin'];

export const crearUsuario = async (req: Request, res: Response, next: NextFunction) => {
  try {
    const { email, password, role, nombre } = req.body || {};

    if (!email || !password || !role || !nombre) {
      return res.status(400).json({ error: 'Todos los campos son obligatorios' });
    }

    if (!ROLES_PERMITIDOS.includes(role)) {
      return res.status(400).json({ error: 'Rol inválido' });
    }

    const emailNormalizado = email.trim().toLowerCase();
    const existingUser = await UserModel.findOne({ email: emailNormalizado });
    if (existingUser) {
      return res.status(400).json({ error: 'El email ya se encuentra registrado' });
    }

    const saltRounds = 10;
    const hashedPassword = await bcrypt.hash(password, saltRounds);

    const newUser = await UserModel.create({
      email: emailNormalizado,
      password: hashedPassword,
      role,
      nombre: nombre.trim(),
    });

    if (req.user?.id) {
      await registrarAuditoria({
        entidad: 'Usuario' as any,
        entidadId: newUser._id.toString(),
        accion: 'CREACION',
        usuarioId: req.user.id,

        motivo: `Alta de usuario con rol ${newUser.role}`,

      return res.status(400).json({ error: 'Patente, marca, modelo y año son obligatorios' });
    }

    const patenteNormalizada = patente.trim().toUpperCase();
    const existePatente = await VehiculoModel.findOne({ patente: patenteNormalizada });
